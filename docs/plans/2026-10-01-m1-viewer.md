# M1 — Visionneuse : plan d'implémentation

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ouvrir l'application et voir le log 16 en 3D, façon maquette : murs avec jonctions en onglet et ouvertures, sols, lumière douce, caméra orbitale et murs du premier plan abaissés.

**Architecture:**
- La géométrie pure (empreintes de murs en onglet, aires, centroïdes) vit dans `src/model/geometry.gd` et se teste sans scène.
- `ProjectStore` (autoload) charge le projet et émet `project_loaded`.
- `src/world/` construit la scène depuis les données : CSG pour les murs, `ArrayMesh` pour les sols et les dessus de murs.
- `Cutaway` et `OrbitState` sont de la logique pure. `WorldView` et `OrbitCamera` les branchent sur la scène.
- `main.gd` assemble l'environnement, la lumière, la vue et la caméra.

**Tech Stack:** Godot 4.7.2, GDScript, GUT 9.7.1.

**Spec:** `docs/specs/2026-09-30-v1-design.md` (§3, §5, §6 caméra orbitale, §9 plan initial, §10, §11 jalon M1).

## Écarts assumés par rapport à la spec

1. **Tolérance des surfaces (spec §9.3, ±3 %)** :
   - Le total habitable est testé à ±3 %.
   - Chaque pièce est testée à ±10 %. Sur un scan, certaines limites sont ambiguës : la bande entre la salle de bain et la chambre 2, le garde-corps de la terrasse.
2. **Jonctions (spec §5)** : un seul algorithme d'onglet par voisin angulaire pour tous les degrés de jonction, au lieu d'un « recouvrement » pour les T. Il donne exactement « la cloison s'arrête sur la face du mur » (test dédié), avec un seul chemin de code.
3. **Ouvertures (spec §5)** :
   - En M1 : découpes, et vitre pour les fenêtres.
   - Repoussés au jalon M6 (finitions) : vantaux, cadres et arcs d'ouverture des portes.
4. **Séjour et cuisine** forment une seule pièce (`r_sejour`, cuisine ouverte), comparée à 8,30 + 22,80 = 31,10 m². Le sol est entièrement en bois.
5. **Calque** : il est livré comme donnée en M1. Son affichage se fait en caméra plan (jalon M4).
6. **Masquage des murs** : l'abaissement anime `scale.y` du nœud du mur au lieu de reconstruire le CSG.

## Review Focus

```gdscript
	return false


func test_isolated_wall_is_a_rectangle() -> void:
```

`src/model/geometry.gd` :

```gdscript
class_name Geometry
extends RefCounted
## Pure 2D geometry on the floor plane (x, z), in meters.

## Beyond this multiple of the wall thickness a miter turns into a square cut,
## so acute corners do not grow long spikes.
const MITER_LIMIT := 2.0
const _PARALLEL_EPSILON := 1e-6


## Mitred footprint of every wall, keyed by wall id. Each footprint is a
## counter-clockwise quad: start-left, end-left, end-right, start-right.
static func wall_footprints(project: ProjectData) -> Dictionary:
	var corners := {}  # "<wall_id>:<node_id>" -> [left, right] relative to the wall's direction away from that node
	for node in project.nodes:
		_solve_node(project, node, corners)
	var footprints := {}
	for wall in project.walls:
		var at_a: Array = corners.get("%s:%s" % [wall.id, wall.a], [])
		var at_b: Array = corners.get("%s:%s" % [wall.id, wall.b], [])
		if at_a.is_empty() or at_b.is_empty():
			continue
		# Seen from b, "left" is the right-hand side of a->b.
		var quad := PackedVector2Array([at_a[0], at_b[1], at_b[0], at_a[1]])
		if Geometry2D.is_polygon_clockwise(quad):
			quad.reverse()
		footprints[wall.id] = quad
	return footprints


static func polygon_area(points: PackedVector2Array) -> float:
	var twice := 0.0
	for i in points.size():
		var p := points[i]
		var q := points[(i + 1) % points.size()]
		twice += p.x * q.y - q.x * p.y
	return absf(twice) / 2.0


static func polygon_centroid(points: PackedVector2Array) -> Vector2:
	var twice_area := 0.0
	var sum := Vector2.ZERO
	for i in points.size():
		var p := points[i]
		var q := points[(i + 1) % points.size()]
		var cross := p.x * q.y - q.x * p.y
		twice_area += cross
		sum += (p + q) * cross
	if absf(twice_area) < _PARALLEL_EPSILON:
		return points[0] if not points.is_empty() else Vector2.ZERO
	return sum / (3.0 * twice_area)


## The room's floor polygon, through the centres of its nodes.
static func room_polygon(project: ProjectData, room: Room) -> PackedVector2Array:
	var points := PackedVector2Array()
	for node_id in room.node_ids:
		points.append(project.get_plan_node(node_id).pos)
	return points


## Floor area inside the walls: the centre-line polygon minus the half thickness
## of every wall running along one of its edges. Corner overlaps are ignored
## (a few cm² per corner).
static func room_net_area(project: ProjectData, room: Room) -> float:
	var area := polygon_area(room_polygon(project, room))
	var count := room.node_ids.size()
	for i in count:
		var a := room.node_ids[i]
		var b := room.node_ids[(i + 1) % count]
		var wall := project.find_wall_between(a, b)
		if wall != null:
			area -= project.wall_length(wall) * wall.thickness / 2.0
	return area


static func _solve_node(project: ProjectData, node: PlanNode, corners: Dictionary) -> void:
	var ends := []  # {wall, dir, half}
	for wall in project.walls:
		if wall.a != node.id and wall.b != node.id:
			continue
		var other := project.get_plan_node(wall.b if wall.a == node.id else wall.a)
		if other == null or other.pos.is_equal_approx(node.pos):
			continue
		ends.append({"wall": wall, "dir": (other.pos - node.pos).normalized(), "half": wall.thickness / 2.0})
	ends.sort_custom(func(e1, e2): return e1.dir.angle() < e2.dir.angle())
	var count := ends.size()
	var lefts := []
	var rights := []
	for e in ends:
		lefts.append(node.pos + _left_normal(e.dir) * e.half)
		rights.append(node.pos - _left_normal(e.dir) * e.half)
	if count > 1:
		for i in count:
			var j := (i + 1) % count
			var e_i: Dictionary = ends[i]
			var e_j: Dictionary = ends[j]
			var hit = Geometry2D.line_intersects_line(
				node.pos + _left_normal(e_i.dir) * e_i.half, e_i.dir,
				node.pos - _left_normal(e_j.dir) * e_j.half, e_j.dir)
			if hit == null:
				continue  # Collinear neighbours keep their square ends.
			var limit := MITER_LIMIT * 2.0 * maxf(e_i.half, e_j.half)
			if node.pos.distance_to(hit) > limit:
				continue
			lefts[i] = hit
			rights[j] = hit
	for i in count:
		corners["%s:%s" % [ends[i].wall.id, node.id]] = [lefts[i], rights[i]]


## Perpendicular rotated towards increasing angle (counter-clockwise in x/z maths).
static func _left_normal(dir: Vector2) -> Vector2:
	return Vector2(-dir.y, dir.x)
```

```gdscript
extends Node
## Autoload holding the open project. Views never edit the data directly:
## they call the store and react to its signals.

signal project_loaded

var project: ProjectData = ProjectData.new()


## Replaces the open project. On failure the current project is kept and the
## reasons are returned.
func load_from_file(path: String) -> PackedStringArray:
	if not FileAccess.file_exists(path):
		return PackedStringArray(["fichier introuvable : %s" % path])
	var result := ProjectSerializer.from_json(FileAccess.get_file_as_string(path))
	if not result.is_ok():
		return result.errors
	project = result.project
	project_loaded.emit()
	return PackedStringArray()
```

`src/world/floor_builder.gd` :

```gdscript
class_name FloorBuilder
extends RefCounted
## Builds one flat floor mesh per room, tinted by floor kind.

## Below this area (m²) a polygon is treated as degenerate.
const MIN_AREA := 1e-6


static func build(project: ProjectData) -> Node3D:
	var root := Node3D.new()
	root.name = "Floors"
	for room in project.rooms:
		var floor_mesh := MeshInstance3D.new()
		floor_mesh.name = room.id
		floor_mesh.mesh = flat_mesh(Geometry.room_polygon(project, room), 0.0)
		floor_mesh.material_override = Palette.material(Palette.floor_color(room.floor_kind))
		root.add_child(floor_mesh)
	return root


## Horizontal mesh of a simple polygon at `height`, facing up. Empty when the
## polygon cannot be triangulated (self-intersecting or degenerate).
static func flat_mesh(polygon: PackedVector2Array, height: float) -> ArrayMesh:
	var mesh := ArrayMesh.new()
	if polygon.size() < 3 or Geometry.polygon_area(polygon) < MIN_AREA:
		return mesh
	var indices := Geometry2D.triangulate_polygon(polygon)
	if indices.is_empty():
		return mesh
	var vertices := PackedVector3Array()
	var normals := PackedVector3Array()
	for p in polygon:
		vertices.append(Vector3(p.x, height, p.y))
		normals.append(Vector3.UP)
	# Godot treats clockwise triangles as front faces; seen from above that is a
	# negative y cross product.
	var faces := PackedInt32Array()
	for i in range(0, indices.size(), 3):
		var a := vertices[indices[i]]
		var b := vertices[indices[i + 1]]
		var c := vertices[indices[i + 2]]
		if (b - a).cross(c - a).y < 0.0:
			faces.append_array([indices[i], indices[i + 1], indices[i + 2]])
		else:
			faces.append_array([indices[i], indices[i + 2], indices[i + 1]])
	var arrays := []
	arrays.resize(Mesh.ARRAY_MAX)
	arrays[Mesh.ARRAY_VERTEX] = vertices
	arrays[Mesh.ARRAY_NORMAL] = normals
	arrays[Mesh.ARRAY_INDEX] = faces
	mesh.add_surface_from_arrays(Mesh.PRIMITIVE_TRIANGLES, arrays)
	return mesh
```

`src/world/wall_builder.gd` :

```gdscript
class_name WallBuilder
extends RefCounted
## Builds one Node3D per wall: the mitred footprint extruded with CSG, openings
## subtracted, a darker cap on top and glass in the windows.

## Extra depth so opening cutters fully cross mitred wall ends.
const CUTTER_MARGIN := 0.05
const GLASS_THICKNESS := 0.02
const CAP_OFFSET := 0.002


static func build(project: ProjectData) -> Node3D:
	var root := Node3D.new()
	root.name = "Walls"
	var footprints := Geometry.wall_footprints(project)
	var wall_material := Palette.material(Palette.WALL)
	var cap_material := Palette.material(Palette.WALL_TOP)
	var glass_material := Palette.material(Palette.GLASS)
	for wall in project.walls:
		if not footprints.has(wall.id):
			continue
		var openings := project.openings.filter(func(o: Opening) -> bool: return o.wall_id == wall.id)
		root.add_child(_build_wall(project, wall, footprints[wall.id], openings,
			wall_material, cap_material, glass_material))
	return root


static func _build_wall(project: ProjectData, wall: Wall, footprint: PackedVector2Array, openings: Array,
		wall_material: Material, cap_material: Material, glass_material: Material) -> Node3D:
	var node := Node3D.new()
	node.name = wall.id
	var body := CSGPolygon3D.new()
	body.name = "Body"
	body.polygon = footprint
	body.depth = wall.height
	body.material = wall_material
	# CSGPolygon3D extrudes its XY polygon along -Z; this turns (x, y) into the
	# floor plane (x, z) and the extrusion upwards.
	body.rotation_degrees = Vector3(90, 0, 0)
	node.add_child(body)

	var start := project.get_plan_node(wall.a).pos
	var dir := (project.get_plan_node(wall.b).pos - start).normalized()
	var yaw := -dir.angle()
	for opening: Opening in openings:
		var center := start + dir * opening.offset
		var cutter := CSGBox3D.new()
		cutter.name = "Cut_%s" % opening.id
		cutter.operation = CSGShape3D.OPERATION_SUBTRACTION
		cutter.size = Vector3(opening.width, opening.height, wall.thickness + CUTTER_MARGIN * 2.0)
		# The body is rotated, so place the cutter in world space through its transform.
		var world := Transform3D(Basis(Vector3.UP, yaw), Vector3(center.x, opening.sill + opening.height / 2.0, center.y))
		cutter.transform = body.transform.affine_inverse() * world
		body.add_child(cutter)
		if opening.kind == "window":
			node.add_child(_glass(opening, wall, world, glass_material))

	node.add_child(_cap(footprint, wall.height + CAP_OFFSET, cap_material))
	return node


static func _glass(opening: Opening, wall: Wall, world: Transform3D, glass_material: Material) -> MeshInstance3D:
	var glass := MeshInstance3D.new()
	glass.name = "Glass_%s" % opening.id
	var box := BoxMesh.new()
	box.size = Vector3(opening.width, opening.height, GLASS_THICKNESS)
	glass.mesh = box
	glass.material_override = glass_material
	glass.transform = world
	return glass


static func _cap(footprint: PackedVector2Array, height: float, cap_material: Material) -> MeshInstance3D:
	var cap := MeshInstance3D.new()
	cap.name = "Cap"
	cap.mesh = FloorBuilder.flat_mesh(footprint, height)
	cap.material_override = cap_material
	return cap
```

```gdscript
class_name Cutaway
extends RefCounted
## Decides which walls drop to a low height so the orbit camera sees inside.

enum Mode { AUTO, FULL, ALL_LOWERED }

const LOWERED_HEIGHT := 0.20
## A wall facing the camera within this angle is lowered.
const MAX_FACING_ANGLE_DEG := 60.0


## `camera_forward` is the camera's horizontal viewing direction on the floor plane.
static func is_lowered(wall_start: Vector2, wall_end: Vector2, apartment_center: Vector2,
		camera_forward: Vector2) -> bool:
	var towards_camera := -camera_forward.normalized()
	var middle := (wall_start + wall_end) / 2.0
	if (middle - apartment_center).dot(towards_camera) <= 0.0:
		return false
	var along := (wall_end - wall_start).normalized()
	var normal := Vector2(-along.y, along.x)
	return absf(normal.dot(towards_camera)) >= cos(deg_to_rad(MAX_FACING_ANGLE_DEG))


## Vertical scale for a wall of `wall_height` in the given mode.
static func target_scale(mode: Mode, lowered: bool, wall_height: float) -> float:
	match mode:
		Mode.FULL:
			return 1.0
		Mode.ALL_LOWERED:
			return LOWERED_HEIGHT / wall_height
		_:
			return LOWERED_HEIGHT / wall_height if lowered else 1.0


static func apartment_center(project: ProjectData) -> Vector2:
	if project.nodes.is_empty():
		return Vector2.ZERO
	var bounds := Rect2(project.nodes[0].pos, Vector2.ZERO)
	for node in project.nodes:
		bounds = bounds.expand(node.pos)
	return bounds.get_center()
```

```gdscript
class_name OrbitState
extends RefCounted
## Orbit camera parameters and their limits, independent of input and nodes.

const MIN_PITCH_DEG := 15.0
const MAX_PITCH_DEG := 85.0
const MIN_DISTANCE := 2.0
const MAX_DISTANCE := 40.0
const ZOOM_FACTOR := 1.1

var target := Vector3.ZERO
var yaw_deg := 30.0
var pitch_deg := 50.0
var distance := 11.0


func rotate(delta_yaw_deg: float, delta_pitch_deg: float) -> void:
	yaw_deg = fposmod(yaw_deg + delta_yaw_deg, 360.0)
	pitch_deg = clampf(pitch_deg + delta_pitch_deg, MIN_PITCH_DEG, MAX_PITCH_DEG)


## Positive steps zoom in.
func zoom(steps: float) -> void:
	distance = clampf(distance / pow(ZOOM_FACTOR, steps), MIN_DISTANCE, MAX_DISTANCE)


## Moves the target in the camera's horizontal frame: x to the right, y forward.
func pan(offset: Vector2) -> void:
	var yaw := deg_to_rad(yaw_deg)
	var right := Vector3(cos(yaw), 0, -sin(yaw))
	var forward := Vector3(-sin(yaw), 0, -cos(yaw))
	target += right * offset.x + forward * offset.y


## Horizontal viewing direction on the floor plane (x, z).
func forward_2d() -> Vector2:
	var yaw := deg_to_rad(yaw_deg)
	return Vector2(-sin(yaw), -cos(yaw))


func camera_transform() -> Transform3D:
	var basis := Basis.from_euler(Vector3(deg_to_rad(-pitch_deg), deg_to_rad(yaw_deg), 0))
	var origin := target + basis.z * distance
	return Transform3D(basis, origin)
```

```gdscript
class_name WorldView
extends Node3D
## Rebuilds walls and floors when a project is loaded, and lowers the walls
## that hide the inside from the orbit camera.

const CUTAWAY_TWEEN_SECONDS := 0.25

var cutaway_mode := Cutaway.Mode.AUTO
var _project: ProjectData
var _walls: Node3D
var _floors: Node3D
var _center := Vector2.ZERO
var _last_forward := Vector2.ZERO
var _lowered := {}  # wall id -> bool, to tween only on change


func show_project(project: ProjectData) -> void:
	_project = project
	for child in get_children():
		remove_child(child)
		child.queue_free()
	_floors = FloorBuilder.build(project)
	_walls = WallBuilder.build(project)
	add_child(_floors)
	add_child(_walls)
	_center = Cutaway.apartment_center(project)
	_lowered.clear()
	_last_forward = Vector2.ZERO


func cycle_cutaway_mode() -> void:
	cutaway_mode = (cutaway_mode + 1) % Cutaway.Mode.size() as Cutaway.Mode
	_lowered.clear()
	_last_forward = Vector2.ZERO


## Called every frame by the camera rig with its horizontal viewing direction.
func update_cutaway(camera_forward: Vector2) -> void:
	if _project == null or camera_forward.is_equal_approx(_last_forward):
		return
	_last_forward = camera_forward
	for wall in _project.walls:
		var node := _walls.get_node_or_null(NodePath(wall.id)) as Node3D
		if node == null:
			continue
		var lowered := Cutaway.is_lowered(_project.get_plan_node(wall.a).pos,
			_project.get_plan_node(wall.b).pos, _center, camera_forward)
		if _lowered.get(wall.id) == lowered:
			continue
		_lowered[wall.id] = lowered
		var target := Cutaway.target_scale(cutaway_mode, lowered, wall.height)
		create_tween().tween_property(node, "scale:y", target, CUTAWAY_TWEEN_SECONDS)
```

`src/cameras/orbit_camera.gd` :

```gdscript
class_name OrbitCamera
extends Camera3D
## Orbit camera: right drag rotates, middle drag pans, wheel zooms,
## double click on a floor centres the view on that room.

const ROTATE_DEG_PER_PX := 0.3
const PAN_M_PER_PX_PER_M := 0.0015  # scaled by distance so panning feels constant
const ZOOM_STEP := 1.0
const FOCUS_SECONDS := 0.4

## Receives the horizontal viewing direction every frame (WorldView.update_cutaway).
signal heading_changed(forward: Vector2)

var orbit := OrbitState.new()
var _project: ProjectData


func frame_project(project: ProjectData) -> void:
	_project = project
	var center := Cutaway.apartment_center(project)
	orbit.target = Vector3(center.x, 0, center.y)
	_apply()


func _ready() -> void:
	_apply()


func _process(_delta: float) -> void:
	heading_changed.emit(orbit.forward_2d())


func _unhandled_input(event: InputEvent) -> void:
	if event is InputEventMouseMotion:
		if event.button_mask & MOUSE_BUTTON_MASK_RIGHT:
			orbit.rotate(-event.relative.x * ROTATE_DEG_PER_PX, -event.relative.y * ROTATE_DEG_PER_PX)
			_apply()
		elif event.button_mask & MOUSE_BUTTON_MASK_MIDDLE:
			var scale := PAN_M_PER_PX_PER_M * orbit.distance
			orbit.pan(Vector2(-event.relative.x, event.relative.y) * scale)
			_apply()
	elif event is InputEventMouseButton and event.pressed:
		match event.button_index:
			MOUSE_BUTTON_WHEEL_UP:
				orbit.zoom(ZOOM_STEP)
				_apply()
			MOUSE_BUTTON_WHEEL_DOWN:
				orbit.zoom(-ZOOM_STEP)
				_apply()
			MOUSE_BUTTON_LEFT:
				if event.double_click:
					_focus_room_under(event.position)


func _focus_room_under(screen_point: Vector2) -> void:
	if _project == null:
		return
	var hit = Plane(Vector3.UP, 0).intersects_ray(project_ray_origin(screen_point), project_ray_normal(screen_point))
	if hit == null:
		return
	var point := Vector2(hit.x, hit.z)
	for room in _project.rooms:
		var polygon := Geometry.room_polygon(_project, room)
		if Geometry2D.is_point_in_polygon(point, polygon):
			var c := Geometry.polygon_centroid(polygon)
			create_tween().tween_method(_set_target, orbit.target, Vector3(c.x, 0, c.y), FOCUS_SECONDS)
			return


func _set_target(value: Vector3) -> void:
	orbit.target = value
	_apply()


func _apply() -> void:
	transform = orbit.camera_transform()
```
